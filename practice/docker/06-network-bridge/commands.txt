docker network create mynet

docker run -d --name web-static --network mynet nginx

cd flask_project
docker build -t flask-api .
docker run -d --name web-api --network mynet -p 5000:5000 flask-api

docker inspect -f '{{range .NetworkSettings.Networks}}{{.IPAddress}}{{end}}' web-api

docker exec -it web-static bash
curl http://web-api:5000/api/data

docker stop web-static web-api
docker rm web-static web-api
